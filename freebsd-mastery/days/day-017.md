### Structure de base (30 min)

```c
/* minishell.c — mini-shell éducatif */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
#include <errno.h>
#include <fcntl.h>

#define MAX_ARGS 64
#define MAX_LINE 1024

/* Parse une ligne en tokens séparés par des espaces */
static int parse_line(char *line, char **argv, int max_args) {
    int argc = 0;
    char *tok = strtok(line, " \t\n");
    while (tok && argc < max_args - 1) {
        argv[argc++] = tok;
        tok = strtok(NULL, " \t\n");
    }
    argv[argc] = NULL;
    return argc;
}

/* Gère les commandes internes (cd, exit) */
static int handle_builtin(char **argv) {
    if (!argv[0]) return 0;
    if (strcmp(argv[0], "exit") == 0) exit(0);
    if (strcmp(argv[0], "cd") == 0) {
        const char *dir = argv[1] ? argv[1] : getenv("HOME");
        if (chdir(dir) < 0) perror("cd");
        return 1;
    }
    return 0; /* pas une commande interne */
}

int main(void) {
    char line[MAX_LINE];
    char *argv[MAX_ARGS];

    while (1) {
        printf("$ ");
        fflush(stdout);

        if (!fgets(line, sizeof(line), stdin)) break; /* EOF (Ctrl-D) */

        int argc = parse_line(line, argv, MAX_ARGS);
        if (argc == 0) continue;
        if (handle_builtin(argv)) continue;

        /* TODO Jour 17 : fork/exec/wait */
        pid_t pid = fork();
        if (pid < 0) { perror("fork"); continue; }
        if (pid == 0) {
            execvp(argv[0], argv);
            perror(argv[0]);
            exit(127);
        }
        /* Père : attend */
        int status;
        while (waitpid(pid, &status, 0) < 0 && errno == EINTR)
            ; /* boucle sur EINTR */
    }
    return 0;
}
```

```sh
$ cd /tmp
$ pwd
$ exit
```

**Critère minimal** : `ls`, `pwd`, `cd`, `exit` fonctionnent.

### Ajout : redirection stdout (10 min)

Avant `execvp()`, ajoute la détection de `>` dans les arguments :

```c
/* Cherche '>' dans argv, redirige stdout */
for (int i = 0; argv[i]; i++) {
    if (strcmp(argv[i], ">") == 0 && argv[i+1]) {
        int fd = open(argv[i+1], O_WRONLY|O_CREAT|O_TRUNC, 0644);
        if (fd < 0) { perror(argv[i+1]); exit(1); }
        dup2(fd, STDOUT_FILENO);
        close(fd);
        argv[i] = NULL; /* tronque argv ici */
        break;
    }
}
```


**Critère** : la redirection `>` fonctionne.
